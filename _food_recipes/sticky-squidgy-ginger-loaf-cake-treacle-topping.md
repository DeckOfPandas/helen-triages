method_groups:
  - name: "Grind the spice blend"
    steps:

  - name: "Make the cake"
    steps:
      - step: "Stir the treacle, muscovado sugar and butter over low heat until melted and smooth. Don't let it boil."

  - name: "Make the treacle topping"
    steps:

      - "Cool slightly."
  - name: "Top the cake"
    steps:
      - "Stab the cake downwards all over with a thin skewer, then dribble the treacle topping over, spreading gently with a knife to encourage it into the holes."
